56[[nodiscard]] constexpr std::size_t digitCount(T value) noexcept {
57 using V = std::remove_cv_t<T>;
58 using Limits = std::numeric_limits<V>;
59
60 if constexpr (std::same_as<V, bool>) {
61 return 1;
62 }
63 else {
64 V x = value;
65
66 // Normalize signed values to negative.
67 if constexpr (Limits::is_signed) {
68 if (x > V{0}) {
69 x = -x;
70 }
71 }
72
73 // Determine the maximum decimal digit count
74 // entirely at compile time.
75 constexpr std::size_t maxDigits = [] {
76 V bound = [] {
77 if constexpr (Limits::is_signed) {
78 return Limits::lowest();
79 }
80 else {
81 return Limits::max();
82 }
83 }();
84
85 std::size_t n = 1;
86
87 while (( bound /= 10 ) != 0) {
88 ++n;
89 }
90
91 return n;
92 }();
93
94 // Generate a binary search tree at compile time.
95 const auto search =
96 [&]<std::size_t lo, std::size_t hi>(
97 auto&& self
98 ) -> std::size_t {
99 if constexpr (lo == hi) {
100 return lo;
101 }
102 else {
103 constexpr std::size_t mid =
104 lo + ( hi - lo + 1 ) / 2;
105
106 // Generate the threshold at compile time.
107 constexpr V threshold = [](std::size_t digits) {
108 V power = V{1};
109
110 if constexpr (Limits::is_signed) {
111 power = -power;
112 }
113
114 for (std::size_t i = 1; i < digits; ++i) {
115 power = static_cast<V>(power * 10);
116 }
117
118 return power;
119 }(mid);
120
121 if constexpr (Limits::is_signed) {
122 return x <= threshold
123 ? self.template operator()<mid, hi>(self)
124 : self.template operator()<lo, mid - 1>(self);
125 }
126 else {
127 return x >= threshold
128 ? self.template operator()<mid, hi>(self)
129 : self.template operator()<lo, mid - 1>(self);
130 }
131 }
132 };
133
134 return search.template operator()<1, maxDigits>(search);
135 }
136}

147template <bool inclusive = true, typename T, typename Y>
148constexpr bool isInRange(const T& low, const T& high, const Y& val) {
149 if constexpr (inclusive) {
150 if (val < low || val > high) {
151 return false;
152 }
153 }
154 else {
155 if (val <= low || val >= high) {
156 return false;
157 }
158 }
159 return true;
160}

175constexpr std::string_view trimWhitespace(const std::string_view str) noexcept {
176 //constexpr std::string_view whitespace = " \t\n\r\f\v";
177 constexpr std::string_view whitespace = " \t";
178
179 const auto first = str.find_first_not_of(whitespace);
180 if (first == std::string_view::npos)
181 return {};
182
183 const auto last = str.find_last_not_of(whitespace);
184 return str.substr(first, last - first + 1);

228inline StrValidation strIsNum(std::string_view str) noexcept {
229 StrValidation result;
230
231 str = trimWhitespace(str);
232
233 // empty string, for some reason
234 if (str.empty()) {
235 return {.status = StrValidation::ValStatus::Empty};
236 }
237
238 //assume that the string is at least 1 non-whitespace char
239 std::size_t i = 0;
240
241 // sign work
242 bool metPeriod = false;
243
244 switch (str[i]) {
245 case '-':
246 result.negative = true;
247 [[fallthrough]];
248 case '+':
249 if (str.size() == 1) {
250 return {.status = StrValidation::ValStatus::Invalid};
251 }
252 str.remove_prefix(1); // move the starting char to the next, if it's a sign
253 break;
254
255 default:
256 break;
257 }
258
259 //integer part
260 for (; i < str.size(); i++) {
261 if (!charIsNum(str[i])) {
262 if (str[i] == '.') {
263 metPeriod = true;
264 break;
265 }
266 return {.status = StrValidation::ValStatus::Invalid};
267 }
268 }
269
270 result.intPart = str.substr(0, i);
271 if (metPeriod) {
272 result.decPart = str.substr(++i);
273 }
274
275 //decimal part
276 for (; i < str.size(); i++) {
277 if (!charIsNum(str[i])) {
278 return {.status = StrValidation::ValStatus::Invalid};
279 }
280 if (!result.hasNonZeroFraction) {
281 if (str[i] > '0') {
282 // literally if larger than 0 (now constrained to 0->9)
283 result.hasNonZeroFraction = true;
284 }
285 }
286 }
287
288 // check if we even scanned any actual digits
289 if (result.intPart.empty() && result.decPart.empty()) {
290 return {.status = StrValidation::ValStatus::Invalid};
291 }
292
293 result.status = StrValidation::ValStatus::Valid;
294
295 return result;
296}

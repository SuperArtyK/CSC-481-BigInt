17TEST(BigintStringConversion, ToStringDecExactValues) {
18 struct Case {
19 std::initializer_list<limb_t> limbs;
20 bool negative;
21 std::string expected;
22 };
23 const Case cases[]{
24 {{0}, false, "0"},
25 {{1}, false, "1"},
26 {{9}, false, "9"},
27 {{10}, false, "10"},
28 {{99}, false, "99"},
29 {{100}, false, "100"},
30 {{STORAGE_MAX}, false, std::string(18, '9')},
31 {{STORAGE_MAX}, true, "-" + std::string(18, '9')},
32 {{1}, true, "-1"},
33 {{42}, true, "-42"},
34 {{0, 1}, false, "1" + std::string(18, '0')},
35 {{1, 1}, false, "1" + std::string(17, '0') + "1"},
36 {{42, 1}, false, "1" + std::string(16, '0') + "42"},
37 {
38 {1, 99999999999999999ULL},
39 false,
40 "99999999999999999" + std::string(17, '0') + "1"
41 },
42 {{0, 10}, false, "10" + std::string(18, '0')},
43 {{STORAGE_MAX, 1}, false, "1" + std::string(18, '9')},
44 {
45 {123456789012345678ULL, 987654321098765432ULL},
46 false,
47 "987654321098765432123456789012345678"
48 },
49 {{0, 0, 1}, false, "1" + std::string(36, '0')},
50 {
51 {123456789012345678ULL, 0, 987654321098765432ULL},
52 true,

129TEST(BigintStringConversion, ToStringDecRejectsNonNormalisedRepresentation) {
130 struct Case {
131 std::initializer_list<limb_t> limbs;
132 bool negative;
133 std::string expectedAfterNormalisation;
134 };
135 const Case cases[]{
136 {{}, false, "0"},
137 {{}, true, "0"},
138 {{0}, true, "0"},
139 {{0, 0}, false, "0"},
140 {{0, 0}, true, "0"},
141 {{5, 0}, false, "5"},
142 {{5, 0}, true, "-5"},
143 {{10, 0, 0}, true, "-10"},
144 };
145
146 for (const Case& testCase : cases) {
147 SCOPED_TRACE(
148 ::testing::Message()
149 << "Input limbs: "
150 << ::testing::PrintToString(std::vector<limb_t>(testCase.limbs))
151 << ", negative: " << testCase.negative
152 );
153
154 BigInt value;
155 tester::load(value, testCase.limbs, testCase.negative);
156 ASSERT_FALSE(value.isNormalised());
157
158 const BigInt& constantValue = value;
159 EXPECT_THROW(constantValue.toStringDec(), std::logic_error);
160 expectRepresentation(
161 value,
162 std::vector<limb_t>(testCase.limbs),
163 testCase.negative
164 );
165
166 // Formatting does not repair the representation; the caller must.
167 value.normalise();
168 ASSERT_TRUE(value.isNormalised());
169 EXPECT_EQ(constantValue.toStringDec(), testCase.expectedAfterNormalisation);
170 }
171}

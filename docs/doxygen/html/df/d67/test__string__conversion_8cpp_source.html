15TEST(BigintStringConversion, ToStringDecExactValues) {
16 struct Case {
17 std::initializer_list<limb_t> limbs;
18 bool negative;
19 std::string expected;
20 };
21 const Case cases[]{
22 {{0}, false, "0"},
23 {{1}, false, "1"},
24 {{9}, false, "9"},
25 {{10}, false, "10"},
26 {{99}, false, "99"},
27 {{100}, false, "100"},
28 {{STORAGE_MAX}, false, std::string(18, '9')},
29 {{STORAGE_MAX}, true, "-" + std::string(18, '9')},
30 {{1}, true, "-1"},
31 {{42}, true, "-42"},
32 {{0, 1}, false, "1" + std::string(18, '0')},
33 {{1, 1}, false, "1" + std::string(17, '0') + "1"},
34 {{42, 1}, false, "1" + std::string(16, '0') + "42"},
35 {
36 {1, 99999999999999999ULL},
37 false,
38 "99999999999999999" + std::string(17, '0') + "1"
39 },
40 {{0, 10}, false, "10" + std::string(18, '0')},
41 {{STORAGE_MAX, 1}, false, "1" + std::string(18, '9')},
42 {
43 {123456789012345678ULL, 987654321098765432ULL},
44 false,
45 "987654321098765432123456789012345678"
46 },
47 {{0, 0, 1}, false, "1" + std::string(36, '0')},
48 {
49 {123456789012345678ULL, 0, 987654321098765432ULL},
50 true,
51 "-987654321098765432"
52 + std::string(18, '0')
53 + "123456789012345678"
54 },
55 {{STORAGE_MAX, STORAGE_MAX, STORAGE_MAX}, false, std::string(54, '9')},
56 };
57
58 for (const Case& testCase : cases) {
59 SCOPED_TRACE(
60 ::testing::Message()
61 << "Input limbs: "
62 << ::testing::PrintToString(std::vector<limb_t>(testCase.limbs))
63 << ", negative: " << testCase.negative
64 );
65
66 BigInt value;
67 tester::load(value, testCase.limbs, testCase.negative);
68 const BigInt& constantValue = value;
69
70 EXPECT_EQ(constantValue.toStringDec(), testCase.expected);
71 expectRepresentation(
72 value,
73 std::vector<limb_t>(testCase.limbs),
74 testCase.negative
75 );
76 }
77}

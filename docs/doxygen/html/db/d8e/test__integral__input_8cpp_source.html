20std::string builtinIntegerText(const T input) {

46TYPED_TEST(BigintIntegralInputTyped, ConstructionAndAssignmentPreserveExactValue) {
47 using T = TypeParam;

50 std::numeric_limits<T>::max()
51 };
52
53 for (const T input : cases) {

18// Independent decimal arithmetic for powers and floating-point maxima.
19// Only integer digits are used: no floating-point formatting or BigInt code.
20void multiplyDecimal(std::string& digits, const int factor) {
21 int carry = 0;
22 for (std::size_t i = digits.size(); i-- > 0;) {
23 const int product = (digits[i] - '0') * factor + carry;
24 digits[i] = static_cast<char>('0' + product % 10);
25 carry = product / 10;
26 }
27 while (carry != 0) {
28 digits.insert(digits.begin(), static_cast<char>('0' + carry % 10));
29 carry /= 10;
30 }
31}

33void decrementDecimal(std::string& digits) {
34 for (std::size_t i = digits.size(); i-- > 0;) {
35 if (digits[i] != '0') {
36 --digits[i];
37 break;
38 }
39 digits[i] = '9';
40 }
41 if (digits.front() == '0' && digits.size() > 1) {
42 digits.erase(0, 1);
43 }
44}

46std::string decimalPower(const int radix, const int exponent) {
47 std::string digits = "1";
48 for (int i = 0; i < exponent; ++i) {
49 multiplyDecimal(digits, radix);
50 }
51 return digits;
52}

54// A maximum finite radix-r floating-point value is
55// (r^digits - 1) * r^(max_exponent - digits).
56// Return its integer part without invoking std::to_chars or BigInt.
57template <typename T>
58std::string maximumFiniteIntegerText() {
59 using Limits = std::numeric_limits<T>;
60 static_assert(Limits::radix >= 2);
61
62 if constexpr (Limits::max_exponent <= 0) {
63 return "0";
64 } else if constexpr (Limits::max_exponent < Limits::digits) {
65 // Maximum lies strictly between r^E - 1 and r^E.
66 std::string digits = decimalPower(Limits::radix, Limits::max_exponent);
67 decrementDecimal(digits);
68 return digits;
69 } else {
70 std::string digits = decimalPower(Limits::radix, Limits::digits);
71 decrementDecimal(digits);
72 for (int i = Limits::digits; i < Limits::max_exponent; ++i) {
73 multiplyDecimal(digits, Limits::radix);
74 }
75 return digits;
76 }
77}

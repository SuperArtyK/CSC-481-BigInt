42template <std::integral T>
43void BigInt::copyFromInt(const T num) {
44 //clear the bigint, as always
45 this->clear(false);
46
47 // Riiiight, handling bools with unsigned is ill-formed
48 // I need to handle them separately
49 // don't forget the "remove const-volatile" or it won't proceed!
50 if constexpr (std::is_same<typename std::remove_cv<T>::type, bool>::value) {
51 this->m_lstDigits.push_back(num);
52 }
53 else {
54 //'kay, actual logic is here
55
56 //unsigned type to handle all
57 using unsigned_t = std::make_unsigned<T>::type;
58
59 unsigned_t unum;
60
61 // just in case the value could be negative
62 if constexpr (std::is_signed<T>::value) {
63 this->m_bNegative = ( num < 0 );
64 unum = ( num < 0 ) ? unsigned_t(0) - static_cast<unsigned_t>(num) : static_cast<unsigned_t>(num);
65 }
66 else {
67 unum = static_cast<unsigned_t>(num);
68 }
69
70 // and if it's bigger than a single limb
71 if constexpr (std::numeric_limits<T>::max() > STORAGE_MAX) {
72 if (unum > STORAGE_MAX) {
73 while (unum > 0) {
74 this->m_lstDigits.push_back(unum % STORAGE_MASK);
75 unum /= STORAGE_MASK;
76 }
77 }
78 else {
79 //awww, I don't wanna repeat codeee
80 this->m_lstDigits.push_back(unum);
81 }
82 }
83 else {
84 this->m_lstDigits.push_back(unum);
85 }
86 }
87}

89template <std::floating_point T>
90void BigInt::copyFromFloat(const T num) {
91 if (!std::isfinite(num)) {
92 throw std::domain_error("Non-finite floating-point value is not permitted");
93 }
94
95 // float is below magnitude 1, so round down
96 if (isInRange<false, T, T>(-1, 1, num)) {
97 this->setToZero();
98 return;
99 }
100
101 //alloc the stack space for char array
102 // YES, we just simply convert the float to a string lmao
103 char fltnum[std::numeric_limits<T>::max_exponent10 + 3]{};
104
105 // to the next one digging this code
106 // Do not forget the formatting rules and precision!!
107 StrValidation result{
108 .intPart = std::string_view(fltnum, std::to_chars(fltnum, fltnum + sizeof( fltnum ) - 1,
109 std::trunc(std::abs(num)), std::chars_format::fixed, 0).ptr),
110 .status = StrValidation::ValStatus::Valid,
111 .negative = ( num < 0 )
112 };
113
114 cdbg << "Converted the passed float to the string and prepared the \"validation result\"" << NLC;
115 this->copyFromValidString(result);
116}

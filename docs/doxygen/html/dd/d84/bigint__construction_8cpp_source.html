73void BigInt::copyFromValidString(const StrValidation& result) {
74 cdbg << "Copying values from the validated string." << NLC;
75 cdbg << "Integer part size: " << result.intPart.size() << "; Fraction part size:" << result.decPart.size()
76 << "; Negative: " << boolToString(result.negative) << "; Fractional part is non-zero: " << boolToString(
77 result.hasNonZeroFraction) << NLC;
78
79 if (result.status == StrValidation::ValStatus::Empty) {
80 cdbg << "Passed string is empty. Setting bigint to 0 instead" << NLC;
81 this->setToZero();
82 return;
83 }
84
85
86 std::string_view str = result.intPart;
87
88 // Remove unnecessary leading zeros.
89 const std::size_t first = str.find_first_not_of('0');
90
91 // Empty integer portion or zero magnitude.
92 if (first == std::string_view::npos) {
93 cdbg << "String is empty after clearing leading 0's. Setting bigint to 0" << NLC;
94 this->setToZero();
95 return;
96 }
97
98 str.remove_prefix(first);
99
100 // Calculate the required number of limbs.
101 const std::size_t limbCount = 1 + ( str.size() - 1 ) / LIMB_DIGITS;
102 cdbg << "Calculated limb count: " << limbCount << NLC;
103
104 // Resize storage once.
105 this->m_lstDigits.resize(limbCount);
106
107 // Convert decimal groups from least to most significant.
108 std::size_t end = str.size();
109
110 for (std::size_t i = 0; i < limbCount; ++i) {
111 const std::size_t start = end > LIMB_DIGITS ? end - LIMB_DIGITS : 0;
112
113 limb_t value = 0;
114
115 for (std::size_t j = start; j < end; ++j) {
116 value = value * 10 + static_cast<limb_t>(str[j] - '0');
117 }
118
119 this->m_lstDigits[i] = value;
120 end = start;
121 }
122
123 // Set the sign after successful magnitude construction.
124 this->m_bNegative = result.negative;
125}

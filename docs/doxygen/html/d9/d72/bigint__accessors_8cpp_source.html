26bool BigInt::setLimbRaw(const std::size_t index, const limb_t value) {
27 cdbg << "Setting limb at index " << index << " to value " << value << NLC;
28 // value is more than what can be stored
29 if constexpr (std::is_signed<limb_t>::value) {
30 if (value < 0) {
31 cdbg << "Value is less than 0 (signed limb_t), failed" << NLC;
32 return false;
33 }
34 }
35
36 if (value > STORAGE_MAX) {
37 cdbg << "Value is more then STORAGE_MAX, failed" << NLC;
38 return false; // failed
39 }
40
41 if (index == std::numeric_limits<std::size_t>::max()) {
42 throw std::length_error("BigInt limb index exceeds maximum storage size");
43 }
44
45 // index is larger than what can be stored
46 if (index >= this->getLimbCount()) {
47 cdbg << "Index is >= than current limb count " << this->getLimbCount() << "; resizing..." << NLC;
48 this->m_lstDigits.resize(index + 1, 0);
49 }
50
51 this->m_lstDigits[index] = value;
52
53 return true; // succeeded
54}

97std::string BigInt::toStringSci(const std::size_t maxSigDig, const bool pad) const {
98 if (!this->isNormalised()) {
99 throw std::logic_error("BigInt must be normalised");
100 }
101
102 if (maxSigDig == 0) {
103 throw std::invalid_argument("Precision must be positive");
104 }
105
106 const std::size_t digitCount = static_cast<std::size_t>(this->getDigitCount());
107 const bool needsRounding = digitCount > maxSigDig;
108 std::string result;
109
110 if (this->isZero() || !needsRounding) {
111 // All digits are needed, so use the existing exact conversion.
112 result = this->toStringDec();
113 }
114 else {
115 // Only the first maxSigDig digits and one rounding digit are needed.
116 // Build that prefix directly from the most significant limbs.
117 if (this->isNegative()) {
118 result.push_back('-');
119 }
120
121 std::size_t remaining = maxSigDig + 1;
122 char numBuf[LIMB_DIGITS + 1];
123 const std::size_t limbCount = this->getLimbCount();
124
125 for (std::size_t i = limbCount; i-- > 0 && remaining > 0;) {
126 const auto end = std::to_chars(
127 numBuf, numBuf + sizeof(numBuf), this->getLimb(i)
128 ).ptr;
129 const std::size_t len = static_cast<std::size_t>(end - numBuf);
130
131 if (i != limbCount - 1) {
132 // Every non-leading limb occupies exactly LIMB_DIGITS digits.
133 const std::size_t zeros = std::min(LIMB_DIGITS - len, remaining);
134 result.append(zeros, '0');
135 remaining -= zeros;
136 }
137
138 const std::size_t copied = std::min(len, remaining);
139 result.append(numBuf, copied);
140 remaining -= copied;
141 }
142 }
143
144 // Special handling for zero.
145 if (result == "0") {
146 if (maxSigDig > 1) {
147 result.push_back('.');
148 result.append(maxSigDig - 1, '0');
149 }
150
151 return result + "e+0";
152 }
153
154 const std::size_t signOffset = this->isNegative() ? 1 : 0;
155 std::size_t exponent = digitCount - 1;
156
157 if (needsRounding) {
158 // Save the first discarded digit before truncating.
159 const char roundingDigit = result[signOffset + maxSigDig];
160 result.resize(signOffset + maxSigDig);
161
162 // Round to nearest, with ties away from zero.
163 if (roundingDigit >= '5') {
164 std::size_t i = result.size();
165
166 while (i > signOffset && result[i - 1] == '9') {
167 result[--i] = '0';
168 }
169
170 if (i == signOffset) {
171 result[signOffset] = '1';
172 ++exponent;
173 }
174 else {
175 ++result[i - 1];
176 }
177 }
178
179 // Strip trailing zeros only when digits were discarded.
180 if (!pad) {
181 while (result.size() > signOffset + 1 && result.back() == '0') {
182 result.pop_back();
183 }
184 }
185 }
186 else if (pad) {
187 result.append(maxSigDig - digitCount, '0');
188 }
189
190 if (result.size() > signOffset + 1) {
191 result.insert(signOffset + 1, 1, '.');
192 }
193
194 result += "e+";
195 result += std::to_string(exponent);
196 return result;
197}

201void BigInt::normalise() {
202 if (this->getLimbCount() == 0) {
203 cdbg << "Bigint has no limbs, setting it to 0" << NLC;
204 this->setToZero();
205 }
206#if DEBUGMODE
207 if (this->getLimbCount() > 1 && this->getLastLimb() == 0) {
208 cdbg << "Non-first limb equals to 0; launching the pop_back loop" << NLC;
209 }
210#endif
211
212
213 while (this->getLimbCount() > 1 && this->getLastLimb() == 0) {
214 this->m_lstDigits.pop_back();
215 }
216
217 if (this->isZero()) {
218 cdbg << "number's magnitude is zero; setting negativity to false (just in case lol)" << NLC;
219 this->setNegative(false);
220 }
221}

26bool BigInt::setLimbRaw(const std::size_t index, const limb_t value) {
27 cdbg<<"Setting limb at index "<<index<<" to value "<<value<<NLC;
28 // value is more than what can be stored
29 if constexpr (std::is_signed<limb_t>::value) {
30 if (value < 0) {
31 cdbg<<"Value is less than 0 (signed limb_t), failed"<<NLC;
32 return false;
33 }
34 }
35
36 if (value > STORAGE_MAX) {
37 cdbg<<"Value is more then STORAGE_MAX, failed"<<NLC;
38 return false; // failed
39 }
40
41 if (index == std::numeric_limits<std::size_t>::max()) {
42 throw std::length_error("BigInt limb index exceeds maximum storage size");
43 }
44
45 // index is larger than what can be stored
46 if (index >= this->getLimbCount()) {
47 cdbg<<"Index is >= than current limb count "<<this->getLimbCount()<<"; resizing..."<<NLC;
48 this->m_lstDigits.resize(index + 1, 0);
49 }
50
51 this->m_lstDigits[index] = value;
52
53 return true; // succeeded
54}

96void BigInt::normalise() {
97 if (this->getLimbCount() == 0) {
98 cdbg<<"Bigint has no limbs, setting it to 0" <<NLC;
99 this->setToZero();
100 }
101#ifdef DEBUGMODE
102 if (this->getLimbCount() > 1 && this->getLastLimb() == 0) {
103 cdbg<<"Non-first limb equals to 0; launching the pop_back loop"<<NLC;
104 }
105#endif
106
107
108 while (this->getLimbCount() > 1 && this->getLastLimb() == 0) {
109 this->m_lstDigits.pop_back();
110 }
111
112 if (this->isZero()) {
113 cdbg<<"number's magnitude is zero; setting negativity to false (just in case lol)"<<NLC;
114 this->setNegative(false);
115 }
116}
